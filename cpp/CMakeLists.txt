cmake_minimum_required(VERSION 3.14)
project(qkernel CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(qkernel qkernel.cpp)
if(MSVC)
  target_compile_options(qkernel PRIVATE $<$<CONFIG:Release>:/O2>)
else()
  target_compile_options(qkernel PRIVATE "$<$<CONFIG:Release>:-O3;-march=native>")
endif()
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(qkernel PUBLIC OpenMP::OpenMP_CXX)
endif()
