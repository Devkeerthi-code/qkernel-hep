cmake_minimum_required(VERSION 3.14)
project(qkernel CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(qkernel qkernel.cpp)
if(MSVC)
  target_compile_options(qkernel PRIVATE $<$<CONFIG:Release>:/O2>)
else()
  target_compile_options(qkernel PRIVATE "$<$<CONFIG:Release>:-O3;-march=native>")
endif()
if(MSVC)
  # the code uses OpenMP >= 3 (collapse, omp simd); MSVC's default /openmp is 2.0 (VS 2019 16.9+)
  target_compile_options(qkernel PRIVATE /openmp:llvm)
elseif(MINGW)
  # self-contained .exe: libgomp, libstdc++ and winpthread linked statically, so no DLLs are needed
  # on PATH at run time (FindOpenMP would link libgomp's import library by full path instead)
  target_compile_options(qkernel PRIVATE -fopenmp)
  target_link_options(qkernel PRIVATE -fopenmp -static)
else()
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(qkernel PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()
